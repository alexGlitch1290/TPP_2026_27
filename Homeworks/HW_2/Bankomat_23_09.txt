using System;
using System.Collections.Generic; 

class Program 
{ 
    static List<string> lst = new List<string>(); 

    static void Balance(decimal a) 
    { 
        Console.WriteLine($"Ваш баланс: {a}"); 
    } 

    static decimal AppBal(decimal a) 
    { 
        Console.WriteLine("Введите сумму для пополнения:");
        decimal am = Convert.ToDecimal(Console.ReadLine()); 
        if (am > 0) 
        { 
            a += am; 
            string rec = $"Пополнение: +{am}"; 
            lst.Add(rec); 
            Console.WriteLine("Счёт успешно пополнен"); 
        } 
        else 
        {
            Console.WriteLine("Ошибка, число должно быть больше 0"); 
        }
        return a; 
    } 

    static decimal Withdraw(decimal a) 
    { 
        Console.WriteLine("Введите сумму для снятия:");
        decimal am = Convert.ToDecimal(Console.ReadLine()); 
        if (am > 0 && am <= a) 
        { 
            a -= am; 
            string rec = $"Снятие средств: -{am}"; 
            lst.Add(rec); 
            Console.WriteLine("Снятие прошло успешно"); 
        } 
        else 
        {
            Console.WriteLine("Ваше число должно быть больше нуля и не превышать ваш баланс"); 
        }
        return a; 
    } 

    static void History() 
    { 
        if (lst.Count == 0) 
        { 
            Console.WriteLine("Ваша история пуста"); 
        } 
        else 
        { 
            for (int i = 0; i < lst.Count; i++) 
            { 
                Console.WriteLine($"{i + 1}. {lst[i]}"); 
            } 
        } 
    } 

    static void Main(string[] args) 
    { 
        Console.WriteLine("Введите начальный баланс:"); 
        decimal bal = Convert.ToDecimal(Console.ReadLine()); 
        bool pr = true; 
        
        while (pr) 
        { 
            Console.WriteLine("\nВыберите действие:");
            Console.WriteLine("1. Показать баланс"); 
            Console.WriteLine("2. Пополнить счёт"); 
            Console.WriteLine("3. Снять деньги"); 
            Console.WriteLine("4. Показать историю операций"); 
            Console.WriteLine("5. Выйти"); 
            
            string ch = Console.ReadLine(); 
            Console.WriteLine(" "); 
            
            switch (ch) 
            { 
                case "1": 
                    Balance(bal); 
                    break; 
                case "2": 
                    bal = AppBal(bal); 
                    break; 
                case "3": 
                    bal = Withdraw(bal); 
                    break; 
                case "4": 
                    History(); 
                    break; 
                case "5": 
                    pr = false; 
                    Console.WriteLine("Конец работы"); 
                    break; 
                default: 
                    Console.WriteLine("Введите цифру от 1 до 5"); 
                    break; 
            } 
        } 
    } 
}
